cmake_minimum_required(VERSION 3.16)
project(soy_touch C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE}" lvgl)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../src/touch.c")
  add_executable(soy ../src/touch.c ../src/touch_input.c ../src/display.c ../src/pinch_draw.c)
  target_compile_options(soy PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(soy PRIVATE lvgl m)
  if(PIXMAN_SOURCE AND PIXMAN_BUILD)
    target_include_directories(soy PRIVATE "${PIXMAN_SOURCE}/pixman" "${PIXMAN_BUILD}/pixman" "${LVGL_SOURCE}")
    target_link_libraries(soy PRIVATE "${PIXMAN_BUILD}/pixman/libpixman-1.a")
    target_compile_definitions(soy PRIVATE SOY_PIXMAN=1)
  endif()
endif()
